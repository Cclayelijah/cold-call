/**
 * Editor entry point — no web app deploy required.
 * Creates OUTREACH_SECRET in Script Properties if missing (also usable later by zoho-upload.html).
 */
function runDay8PartnerOutreachUpload() {
  var props = PropertiesService.getScriptProperties();
  var secret = props.getProperty('OUTREACH_SECRET');
  if (!secret) {
    secret = Utilities.getUuid().replace(/-/g, '') + Utilities.getUuid().replace(/-/g, '');
    props.setProperty('OUTREACH_SECRET', secret);
    Logger.log('Created OUTREACH_SECRET in Script Properties (keep for web upload page).');
  }
  if (typeof upsertPartnerOutreachLeads !== 'function') {
    throw new Error('Add PartnerOutreach.gs to this project first (function upsertPartnerOutreachLeads missing).');
  }
  var result = upsertPartnerOutreachLeads({ secret: secret, leads: DAY8_PARTNER_LEADS });
  Logger.log(JSON.stringify(result, null, 2));
  if (!result || !result.ok) {
    throw new Error('Day 8 upsert incomplete: ' + JSON.stringify(result));
  }
  Logger.log('Day 8 saved ' + result.saved + '/' + result.total + ' Partner Outreach leads.');
  return result;
}
